Supplementary Methods

A global cross-species pathogen network reveals the evolutionary and transmission potential of the Rhinolophus sinicus gut microbiome
Computational workflow and analysis scripts

Computational methods. This document presents the analysis scripts and command templates in the order of the major analytical stages. The parameters are included in the corresponding code blocks. To execute an individual script, save its code using the filename shown above the block, install the required software, and specify the relevant input and output paths and reference files. For scripts using environment variables, set the required variables before execution.

The code is intended for a Linux environment with Bash and Python. Individual stages require their corresponding bioinformatic tools and Python libraries (including NumPy, Numba, Biopython, pandas and Matplotlib).

S1  Read quality control

Short-read and long-read filtering. Public read quality filtering uses Q20, a maximum low-quality base fraction of 40%, a minimum length of 50 bp and poly-G trimming. The bat short-read command uses the software defaults for quality filtering. For long reads, the command template below uses a minimum read length of 500 bp and a minimum mean base-call accuracy of 80% as proposed filtering settings; the original run-specific thresholds are not available in the archived scripts.

fix_special_fasta_structure() {
    local input_file="$1"
    local output_file="$2"

    echo "Normalizing FASTA headers..."

    awk '
    BEGIN {
        skip_next = 0
        header_printed = 0
    }
    /^>/ {
        if (skip_next) {
            skip_next = 0
            next
        } else {
            current_header = $0
            header_printed = 0
            skip_next = 1
        }
        next
    }
    {
        if (!header_printed && current_header != "") {
            print current_header
            header_printed = 1
        }
        if (header_printed) {
            print $0
        }
    }
    ' "$input_file" > "$output_file"

    echo "Completed: $input_file -> $output_file"
    echo "Original sequence count: $(grep -c "^>" "$input_file")"
    echo "New sequence count: $(grep -c "^>" "$output_file")"
}

mapfile -t samples < "${SAMPLE_LIST:?Set SAMPLE_LIST}"
for sample in "${samples[@]}"; do
    input_file="$(configured_path "${P_INPUT_FILE:?Set P_INPUT_FILE}" "sample" "${sample}")"
    output_file="$(configured_path "${P_OUTPUT_FILE:?Set P_OUTPUT_FILE}" "sample" "${sample}")"

    if [ -f "$input_file" ]; then
        echo "Processing sample: $sample"

        cp "$input_file" "${input_file}.backup"

        fix_special_fasta_structure "$input_file" "$output_file"

        echo "Checking results:"
        echo "First six original lines:"
        head -6 "$input_file"
        echo "First four output lines:"
        head -4 "$output_file"

S3.8  summarize_assembly_qc.py

#!/usr/bin/env python3


import os
import pandas as pd


samples = [line.strip() for line in open(os.environ["SAMPLE_LIST"], encoding="utf-8") if line.strip()]


result_file_template = os.environ["P_RESULT_FILE_TEMPLATE"]

output_dir = os.environ["P_OUTPUT_DIR"]

output_filename = "quast_full_summary.tsv"


os.makedirs(output_dir, exist_ok=True)


all_samples_data = {}

all_metrics = set()


for sample in samples:
    file_path = result_file_template.format(sample=sample)
    sample_data = {}
    
    if not os.path.exists(file_path):
        print(f"Missing QUAST result for {sample}: {file_path}")
        all_samples_data[sample] = sample_data
        continue
    

    with open(file_path, 'r') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            

            if '\t' in line:
                metric, value = line.split('\t', 1)
            else:
                parts = line.split()
                if len(parts) < 2:
                    continue
                value = parts[-1]
                metric = ' '.join(parts[:-1])
            
            metric = metric.strip()
            value = value.strip()
            
            sample_data[metric] = value
            all_metrics.add(metric)
    
    all_samples_data[sample] = sample_data


sorted_metrics = sorted(all_metrics)
summary_data = {}
for sample in samples:
    sample_data = all_samples_data[sample]
    summary_data[sample] = {metric: sample_data.get(metric, "NA") for metric in sorted_metrics}


summary_df = pd.DataFrame(summary_data).T
summary_df.index.name = "Sample"


output_file = os.path.join(output_dir, output_filename)
summary_df.to_csv(output_file, sep='\t')

print(f"Metrics saved to {output_file}")
print(f"Metrics: {len(sorted_metrics)}; samples: {len(samples)}")

        input_id_files+=("$genomad_ids")
        tool_names+=("geNomad")
    else

    fi

        input_id_files+=("$virsorter_full")
        tool_names+=("VirSorter(full)")
    else
        echo "Missing full VirSorter IDs: $virsorter_full" | tee -a "$log_file"

#!/usr/bin/env python3


import argparse
import csv
import math
from Bio import Phylo


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--tree", required=True, help="Newick tree with branch lengths")
    p.add_argument("--pairs", required=True, help="TSV with query and reference columns")
    p.add_argument("--out", required=True)
    a = p.parse_args()
    tree = Phylo.read(a.tree, "newick")
    tips = tree.get_terminals()
    by_name = {x.name: x for x in tips}
    if len(by_name) != len(tips) or None in by_name:
        raise ValueError("Tree tip names must be present and unique")
    for clade in tree.find_clades():
        if clade is tree.root:
            continue
        if clade.branch_length is None or not math.isfinite(clade.branch_length) or clade.branch_length < 0:
            raise ValueError("Missing, negative or invalid branch length")
    results = []
    with open(a.pairs, encoding="utf-8-sig", newline="") as f:
        r = csv.DictReader(f, delimiter="\t")
        if not {"query", "reference"}.issubset(r.fieldnames or []):
            raise ValueError("Pair table requires query and reference columns")
        for row in r:
            q, ref = row["query"], row["reference"]
            if q not in by_name or ref not in by_name or q == ref:
                raise ValueError(f"Invalid query/reference pair: {q}, {ref}")
            results.append((q, ref, tree.distance(by_name[q], by_name[ref])))
    with open(a.out, "w", encoding="utf-8", newline="") as f:
        w = csv.writer(f, delimiter="\t", lineterminator="\n")
        w.writerow(["query", "reference", "patristic_distance"])
        w.writerows(results)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3


import argparse
import csv
import math
import re
from collections import defaultdict, Counter
from pathlib import Path

EXCLUDE = {"unknown", "Autographiviridae", "Arenbergviridae", "Baculoviridae",
           "Caulimoviridae", "Chaseviridae", "Herelleviridae", "Inoviridae",
           "Iridoviridae", "Metaviridae", "Poxviridae", "Retroviridae"}

HEADER = ["\u57fa\u56e0\u540d", "contigID", "\u75c5\u6bd2\u5bb6\u65cf", "\u5c5e", "\u79cd", "\u65b0\u7c7b\u578b", "\u5c5e\u79cd\u5212\u5206", "\u6837\u672c\u6765\u6e90"]
SPECIES, GENUS = "\u5019\u9009\u65b0\u7269\u79cd", "\u5019\u9009\u65b0\u5c5e"


def read_table(path, auto=False):
    with open(path, encoding="utf-8-sig", newline="") as handle:
        first = handle.readline()
        if not first:
            raise ValueError(f"Empty input: {path}")
        handle.seek(0)
        sep = ("," if first.count(",") > first.count("\t") else "\t") if auto else "\t"
        return list(csv.reader(handle, delimiter=sep))


def numeric(s):
    try:
        v = float(s)
        return v if math.isfinite(v) else None
    except (ValueError, TypeError):
        return None


def select(genetic, family, taxonomy, sample, *, confirm_legacy=False):
    header, *rows = read_table(genetic, auto=True)
    if len(header) < 21:
        raise ValueError("Legacy candidate table must have at least 21 columns.")
    print(f"{sample}: columns 8/9/17/18/20/21 = " + " | ".join(header[i-1] for i in (8,9,17,18,20,21)))


    if not confirm_legacy:
        raise ValueError("Verify the printed header against the archived analysis, then use --confirm-legacy-columns.")
    fam = defaultdict(list)
    for r in read_table(family)[1:]:
        if len(r) < 2:
            raise ValueError("Malformed family annotation row")
        if r[-1] not in EXCLUDE:
            fam[r[0]].append(r[-1])
    tax = defaultdict(list)
    for r in read_table(taxonomy)[1:]:
        if len(r) < 3:
            raise ValueError("Malformed taxonomy row")
        tax[r[0]].append((r[-2], r[-1]))
    selected = []
    stats = Counter()
    for r in rows:
        if len(r) < 21:
            raise ValueError(f"Malformed candidate row in {genetic}")
        if r[16] != "\u662f" or r[17] not in (SPECIES, GENUS):
            continue
        sh, uf, value20, value21 = map(numeric, (r[7], r[8], r[19], r[20]))
        if None in (sh, uf, value20, value21):
            stats["nonnumeric_eligible_rows"] += 1
            continue
        if sh < 90 or uf < 95 or value21 < 80:
            continue
        if (r[17] == SPECIES and value20 > 85) or (r[17] == GENUS and value20 > 75):
            continue
        stats["passed_numeric_filters"] += 1
        gene = r[0]
        match = re.search(r"opera_contig_[0-9]+_np[0-9]+", gene) or re.search(r"contig_[0-9]+", gene)
        contig = match.group(0) if match else "unknown"
        if gene not in tax or contig not in fam:
            stats["unmatched_annotation_rows"] += 1
            continue
        for g, s in sorted(tax[gene]):
            for f in sorted(fam[contig]):
                level = "\u79cd" if r[17] == SPECIES else "\u5c5e"
                selected.append([gene, contig, f, g, s, r[17], level, sample])

    selected.sort(key=lambda r: (r[1], r[0], r[5], r[3], r[4], r[2]))
    print(f"{sample}: {len(selected)} joined rows; audit={dict(stats)}")
    return selected


def write_table(path, rows, header):
    with open(path, "w", encoding="utf-8", newline="") as f:
        w = csv.writer(f, delimiter="\t", lineterminator="\n")
        w.writerow(header)
        w.writerows(rows)


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--manifest", required=True, help="TSV columns sample, genetic, family, taxonomy; file paths relative to manifest or absolute")
    p.add_argument("--summary", required=True)
    p.add_argument("--deduplicated", required=True)
    p.add_argument("--confirm-legacy-columns", action="store_true")
    args = p.parse_args()
    manifest = Path(args.manifest).resolve()
    all_rows = []
    seen_samples = set()
    with manifest.open(encoding="utf-8-sig", newline="") as f:
        reader = csv.DictReader(f, delimiter="\t")
        if not {"sample","genetic","family","taxonomy"}.issubset(reader.fieldnames or []):
            raise ValueError("Manifest requires sample, genetic, family, taxonomy columns.")
        for row in reader:
            sample = row["sample"]
            if not sample or sample in seen_samples:
                raise ValueError("Sample IDs must be nonempty and unique.")
            seen_samples.add(sample)
            paths = [manifest.parent / row[key] for key in ("genetic", "family", "taxonomy")]
            all_rows.extend(select(*paths, sample, confirm_legacy=args.confirm_legacy_columns))
    if not seen_samples:
        raise ValueError("Empty sample manifest")
    unique = {}
    for row in all_rows:
        unique.setdefault((row[3], row[4], row[6]), row)
    write_table(args.summary, all_rows, HEADER)
    write_table(args.deduplicated, unique.values(), HEADER)
    print(f"All rows: {len(all_rows)}; unique genus/species/rank keys: {len(unique)}")
    print("Unique counts by rank:", dict(Counter(r[6] for r in unique.values())))


if __name__ == "__main__":
    main()

def lengths(path):
    result={}; name=None

@njit(cache=True)
def curveball_shuffle_inplace(mat, trades):

    nr, nc = mat.shape
    a = np.empty(nc, np.int64)
    b = np.empty(nc, np.int64)
    mix = np.empty(nc, np.int64)
    accepted = 0
    for _ in range(trades):
        i = np.random.randint(nr)
        j = np.random.randint(nr - 1)
        if j >= i:
            j += 1
        n1 = 0
        n2 = 0
        for c in range(nc):
            if mat[i, c] and not mat[j, c]:
                a[n1] = c
                n1 += 1
            elif mat[j, c] and not mat[i, c]:
                b[n2] = c
                n2 += 1
        if n1 == 0 or n2 == 0:
            continue
        accepted += 1
        n = n1 + n2
        for t in range(n1):
            mix[t] = a[t]
        for t in range(n2):
            mix[n1 + t] = b[t]
        for p in range(n - 1, 0, -1):
            q = np.random.randint(p + 1)
            tmp = mix[p]
            mix[p] = mix[q]
            mix[q] = tmp
        for p in range(n):
            c = mix[p]
            if p < n1:
                mat[i, c] = 1
                mat[j, c] = 0
            else:
                mat[i, c] = 0
                mat[j, c] = 1
    return accepted


@njit(cache=True)
def optimize_barber_modularity(M, restarts, seed, maxpasses=40):


    np.random.seed(seed)
    nr, nc = M.shape
    n = nr + nc
    m = np.float64(M.sum())
    rowdeg = M.sum(axis=1)
    coldeg = M.sum(axis=0)
    B = np.empty((nr, nc), np.float64)
    for r in range(nr):
        for c in range(nc):
            B[r, c] = M[r, c] - (rowdeg[r] * coldeg[c]) / m

    scores = np.empty(2*n, np.float64)
    counts = np.empty(2*n, np.int64)
    order = np.arange(n)
    best_q = -2.
    bestlabels = np.empty(n, np.int64)
    for start in range(restarts):
        labels = np.empty(n, np.int64)
        if start == 0:
            for v in range(n):
                labels[v] = v
        else:
            ngroups = max(2, min(n, 8 + (start * 7) % 35))
            for v in range(n):
                labels[v] = np.random.randint(ngroups)
        counts[:] = 0
        for v in range(n):
            counts[labels[v]] += 1

        for _ in range(maxpasses):
            np.random.shuffle(order)
            changed = False
            for x in range(n):
                v = order[x]
                scores[:] = 0.0
                if v < nr:
                    for c in range(nc):
                        scores[labels[nr+c]] += B[v, c]
                    ostart, ostop = nr, n
                else:
                    col = v - nr
                    for r in range(nr):
                        scores[labels[r]] += B[r, col]
                    ostart, ostop = 0, nr
                old = labels[v]
                oldvalue = scores[old]
                best = oldvalue
                chosen = old
                for j in range(ostart, ostop):
                    g = labels[j]
                    if scores[g] > best + 1e-12:
                        best = scores[g]
                        chosen = g


                if best < -1e-12:
                    for g in range(2*n):
                        if counts[g] == 0:
                            chosen = g
                            best = 0.0
                            break
                if chosen != old and best > oldvalue + 1e-12:
                    labels[v] = chosen
                    counts[old] -= 1
                    counts[chosen] += 1
                    changed = True
            if not changed:
                break
        q = 0.
        for r in range(nr):
            for c in range(nc):
                if labels[r] == labels[nr+c]:
                    q += B[r, c]
        q /= m
        if q > best_q:
            best_q = q
            bestlabels[:] = labels
    return best_q, bestlabels


def independent_barber_Q(M, lab):

    nr, nc = M.shape
    k = M.sum(axis=1).astype(float)
    d = M.sum(axis=0).astype(float)
    m = float(M.sum())
    Q = 0.0
    for x in np.unique(lab):
        rows = np.flatnonzero(lab[:nr] == x)
        cols = np.flatnonzero(lab[nr:] == x)
        if len(rows) and len(cols):
            observed = float(M[np.ix_(rows, cols)].sum())
            expected = float(k[rows].sum() * d[cols].sum() / m)
            Q += observed - expected
    return Q / m


def count_switches(M):
    num = 0
    for a in range(len(M)):
        for b in range(a + 1, len(M)):
            a_only = np.count_nonzero((M[a] == 1) & (M[b] == 0))
            b_only = np.count_nonzero((M[a] == 0) & (M[b] == 1))
            num += int(a_only * b_only)
    return num


def read_incidence(source, mode, human_merge=True):
    fields = ('Species', 'Viral_Family') if mode == 'viral' else ('Wild_Animal', 'WHO_Pathogen')
    edges = set()
    human_labels = set()
    raw_hosts = set()
    skipped = 0
    nrows = 0
    with open(source, encoding='utf-8-sig', newline='') as handle:
        reader = csv.DictReader(handle, delimiter='\t')
        required = set(fields) | {'Frequency'}
        if not reader.fieldnames or not required.issubset(reader.fieldnames):
            raise ValueError(f'Expected columns {required}; got {reader.fieldnames}')
        for index, row in enumerate(reader, start=2):
            nrows += 1
            host = (row[fields[0]] or '').strip()
            taxon = (row[fields[1]] or '').strip()
            if not host or not taxon:
                raise ValueError(f'Missing host/taxon on input line {index}')
            raw_hosts.add(host)
            try:
                frequency = float(row['Frequency'])
            except (ValueError, TypeError):
                raise ValueError(f'Invalid Frequency at line {index}: {row["Frequency"]}') from None
            if not math.isfinite(frequency):
                raise ValueError(f'Non-finite Frequency at line {index}')
            if frequency <= 0:
                skipped += 1
                continue
            if human_merge and host.casefold() in {'human', 'homo sapiens'}:
                human_labels.add(host)
                host = 'Homo sapiens'
            edges.add((host, taxon))
    hs = sorted({h for h, _ in edges})
    ts = sorted({t for _, t in edges})
    hidx = {h:i for i,h in enumerate(hs)}
    tidx = {t:i for i,t in enumerate(ts)}
    M = np.zeros((len(hs), len(ts)), dtype=np.uint8)
    for h,t in edges:
        M[hidx[h], tidx[t]] = 1
    if M.shape[0] < 2 or M.shape[1] < 2:
        raise ValueError('Need >=2 hosts and >=2 taxa')
    audit = {'source_rows':nrows, 'skipped_nonpositive_frequency':skipped,
             'raw_host_labels':len(raw_hosts), 'human_labels_merged':sorted(human_labels),
             'hosts_after_cleaning':len(hs), 'taxa':len(ts), 'edges':int(M.sum())}
    return M, hs, ts, audit


def write_matrix(path, M, hs, ts):
    with open(path, 'w', encoding='utf-8', newline='') as f:
        w=csv.writer(f, delimiter='\t')
        w.writerow(['Host'] + ts)
        for i,h in enumerate(hs):
            w.writerow([h] + M[i].tolist())


def save_null_results(path, rows):
    tmp = str(path) + '.tmp'
    with open(tmp, 'w', encoding='utf-8', newline='') as f:
        w = csv.DictWriter(f, fieldnames=['iteration','Q_null','changed_cells','accepted_trades'])
        w.writeheader()
        w.writerows(rows)
    os.replace(tmp, path)


def read_null_results(path):
    if not path.exists():
        return []
    with path.open(encoding='utf-8', newline='') as f:
        out = list(csv.DictReader(f))
    rows = []
    for i, row in enumerate(out, start=1):
        if int(row['iteration']) != i:
            raise ValueError('Checkpoint iteration numbers not consecutive')
        rows.append({'iteration':i, 'Q_null':float(row['Q_null']),
                     'changed_cells':int(row['changed_cells']),
                     'accepted_trades':int(row['accepted_trades'])})
    return rows


def atomic_json(path, obj):
    tmp = str(path) + '.tmp'
    with open(tmp, 'w', encoding='utf-8') as f:
        json.dump(obj, f, ensure_ascii=False, indent=2)
    os.replace(tmp, path)


def run(args):
    start = time.perf_counter()
    source = Path(args.input).resolve()
    if not source.is_file():
        raise FileNotFoundError(source)
    if args.n_null < 2 or args.starts < 1 or args.trades_per_row < 1:
        raise ValueError('--n-null >= 2, --starts >= 1 and --trades-per-row >= 1 required')
    M, hosts, taxa, audit = read_incidence(source, args.mode, not args.no_merge_human)
    out = Path(args.out).resolve()
    out.mkdir(parents=True, exist_ok=True)
    sha = hashlib.sha256(source.read_bytes()).hexdigest()
    run_options = {'mode':args.mode,'input_sha256':sha,'merge_human':not args.no_merge_human,
                   'n_null':args.n_null,'starts':args.starts,
                   'trades_per_row':args.trades_per_row,'seed':args.seed}
    config = out / 'analysis_config.json'
    if config.exists():
        old_config = json.loads(config.read_text(encoding='utf-8'))
        if old_config != run_options:
            raise ValueError('Output folder already contains a DIFFERENT analysis. Use another --out.')
    else:
        atomic_json(config, run_options)
    write_matrix(out/'binary_incidence_matrix.tsv',M,hosts,taxa)
    atomic_json(out/'input_audit.json',audit)
    nr,nc = M.shape
    n_edges = int(M.sum())
    n_switches = count_switches(M)
    print(f'Input: {source.name}',flush=True)
    print(f'Network: hosts={nr}, taxa={nc}, edges={n_edges}, density={M.mean():.5f}, '
          f'valid 2x2 switches={n_switches}',flush=True)
    print('Human merged labels:',audit['human_labels_merged'],flush=True)
    if n_switches == 0:
        atomic_json(out/'summary.json', {'status':'fixed_fixed_degenerate',**audit,
                    'valid_2x2_switches':0,'p_value':None,'z_score':None,
                    'explanation':'No degree-preserving alternatives; inferential null p-value is invalid.'})
        print('DEGENERATE: no fixed-fixed randomization possible; no p-value.', flush=True)
        return

    Qobs, labels = optimize_barber_modularity(M, args.starts, args.seed)
    verify = independent_barber_Q(M,labels)
    if not np.isclose(Qobs,verify,rtol=1e-9,atol=1e-9):
        raise AssertionError(f'Independent modularity check failed: {Qobs} vs {verify}')
    with open(out/'observed_modules.tsv','w',encoding='utf-8',newline='') as f:
        w=csv.writer(f,delimiter='\t')
        w.writerow(['node_type','node','module','degree'])
        for i,h in enumerate(hosts):w.writerow(['host',h,int(labels[i]),int(M[i].sum())])
        for j,t in enumerate(taxa):w.writerow(['taxon',t,int(labels[nr+j]),int(M[:,j].sum())])
    print(f'Q_observed={Qobs:.8f}; independent calculation check=PASS',flush=True)

    checkpoint = out/'null_modularity_values.csv'
    rows = read_null_results(checkpoint)
    if rows:
        if not args.resume:
            raise ValueError('Found prior checkpoint. Use --resume or a new --out directory.')
        if len(rows)>args.n_null:
            raise ValueError('Checkpoint longer than requested simulations')
        print(f'Resuming from {len(rows)} completed null matrices',flush=True)
    elif args.resume:
        print('No prior progress found, starting new run',flush=True)
    original_row_sums=M.sum(axis=1).astype(np.int64)
    original_col_sums=M.sum(axis=0).astype(np.int64)
    trades=args.trades_per_row*nr
    save_every = max(1,args.save_every)
    for i in range(len(rows),args.n_null):
        N=M.copy()
        set_numba_seed(args.seed+100000+i)
        accepted=int(curveball_shuffle_inplace(N,trades))
        if (not np.array_equal(N.sum(axis=0),original_col_sums) or
                not np.array_equal(N.sum(axis=1),original_row_sums)):
            raise AssertionError(f'Degree preservation failed at null {i+1}')
        changed=int(np.count_nonzero(M!=N))
        q, labn = optimize_barber_modularity(N,args.starts,args.seed+200000+i)
        if not np.isclose(q,independent_barber_Q(N,labn),rtol=1e-8,atol=1e-8):
            raise AssertionError(f'Barber Q independent check failed at {i+1}')
        rows.append({'iteration':i+1,'Q_null':float(q),
                     'changed_cells':changed,'accepted_trades':accepted})
        if (i+1)%save_every==0 or i+1==args.n_null:
            save_null_results(checkpoint,rows)
            print(f'Completed {i+1}/{args.n_null} | latest Q={q:.5f} | changed cells={changed} '
                  f'| elapsed={time.perf_counter()-start:.1f}s',flush=True)
    qnull=np.array([r['Q_null'] for r in rows],dtype=float)
    sd=float(qnull.std(ddof=1))
    nge=int(np.count_nonzero(qnull>=Qobs-1e-12))
    p=(nge+1)/(len(rows)+1)
    summary={**audit,'status':'complete',
             'network_density':float(M.mean()),'valid_2x2_switches':n_switches,
             'Q_observed':float(Qobs),'Q_null_mean':float(qnull.mean()),
             'Q_null_sd':sd,'Z':float((Qobs-qnull.mean())/sd) if sd>0 else None,
             'p_upper_empirical':float(p),'n_null_at_or_above_observed':nge,
             'n_null':len(rows),'starts':args.starts,'trades_per_row':args.trades_per_row,
             'min_changed_cells':int(min(r['changed_cells'] for r in rows)),
             'mean_changed_cells':float(np.mean([r['changed_cells'] for r in rows])),
             'null_lag1_autocorrelation':float(np.corrcoef(qnull[:-1],qnull[1:])[0,1]) if len(qnull)>4 and qnull.std()>0 else None,
             'time_seconds':float(time.perf_counter()-start),
             'python_version':platform.python_version(), 'numpy_version':np.__version__}
    import numba
    summary['numba_version']=numba.__version__
    atomic_json(out/'summary.json',summary)
    try:
        import matplotlib
        matplotlib.use('Agg')
        import matplotlib.pyplot as plt
        fig, ax=plt.subplots(figsize=(7.5,4.5))
        ax.hist(qnull,bins=30)
        ax.axvline(Qobs,ls='--',lw=2,label=f'Observed Q = {Qobs:.4f}')
        ax.set_xlabel('Barber bipartite modularity Q')
        ax.set_ylabel('Number of randomized networks')
        ax.legend()
        fig.tight_layout()
        fig.savefig(out/'null_modularity_distribution.png',dpi=220)
        plt.close(fig)
    except ImportError:
        print('Optional matplotlib not available: skipping PNG output',flush=True)
    print(f'FINISHED: Q_obs={Qobs:.6f}, Q_null={qnull.mean():.6f} +/- {sd:.6f}, '
          f'Z={summary["Z"]}, p={p:.6g}',flush=True)
    print('Saved results in:',out,flush=True)


def self_test():

    small=np.array([[1,0],[0,1]], dtype=np.uint8)
    labs=np.array([0,1,0,1],dtype=np.int64)
    assert abs(independent_barber_Q(small,labs)-0.5)<1e-12
    assert abs(independent_barber_Q(np.ones((3,2),dtype=np.uint8),np.zeros(5,dtype=np.int64)))<1e-12
    set_numba_seed(12345)
    x=small.copy()
    curveball_shuffle_inplace(x,50)
    assert np.array_equal(small.sum(0),x.sum(0))
    assert np.array_equal(small.sum(1),x.sum(1))
    q,l=optimize_barber_modularity(small,8,111)
    assert abs(q-independent_barber_Q(small,l))<1e-12

    a=small.copy();b=small.copy()
    set_numba_seed(100);curveball_shuffle_inplace(a,20)
    set_numba_seed(100);curveball_shuffle_inplace(b,20)
    assert np.array_equal(a,b)

    nested=np.array([[1,1,1],[1,1,0],[1,0,0]],dtype=np.uint8)
    assert count_switches(nested)==0
    print('SELF-TEST PASS: modularity formula, margins, seeded randomization, nesting',flush=True)
